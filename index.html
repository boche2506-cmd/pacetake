// 🟢 1. 將登入邏輯綁定到 window 全域，這樣任何地方都叫得到它 window.executeGoogleLogin = async function() { try { // 自動偵測變數到底藏在哪裡 const currentAuth = typeof auth !== 'undefined' ? auth : (window.auth || window.firebaseAuth); const currentProvider = typeof provider !== 'undefined' ? provider : (window.provider || window.googleProvider); const loginFunc = typeof signInWithPopup === 'function' ? signInWithPopup : window.signInWithPopup; if (!currentAuth || !loginFunc) { alert("⚠️ 錯誤：網頁尚未載入 Firebase Auth 模組，請重新整理試試！"); return; } console.log("🚀 開始執行 Google 彈窗登入..."); const result = await loginFunc(currentAuth, currentProvider); // 登入成功後的路由跳轉 if (typeof handleUserSyncAndRoleRouting === 'function') { await handleUserSyncAndRoleRouting(result.user); } else if (window.handleUserSyncAndRoleRouting) { await window.handleUserSyncAndRoleRouting(result.user); } } catch (error) { console.error("Google 登入失敗詳細原因：", error); alert("連線失敗，請檢查網路或 Firebase 彈窗設定！"); } }; // 🟢 2. 乾淨綁定按鈕點擊事件，順便清除舊的監聽打架問題 document.addEventListener('DOMContentLoaded', () => { const gBtn = document.getElementById('googleLoginAction'); if (gBtn) { const newGBtn = gBtn.cloneNode(true); gBtn.parentNode.replaceChild(newGBtn, gBtn); newGBtn.addEventListener('click', window.executeGoogleLogin); } });
